Built for real devices

Health Connect has no push stream. On resume the first aggregate is often 0 until vendor apps finish writing — use useForegroundPoll / pollUntil. 

import {
  Permission,
  ensurePermissions,
  readSteps,
  timeRangeToday,
  useForegroundPoll,
} from '@solutionforest/android-health-connect';

const { value } = useForegroundPoll(
  () => readSteps({ timeRangeFilter: timeRangeToday() }),
  { ready: (r) => (r.records?.length ?? 0) > 0 },
);
